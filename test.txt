# Run on the server, in the checkout that has the updated php-site.conf and
# certbot/domains.txt (sync/deploy the repo before running this).
cd ~/server-setup

# Web root for the site, plus a per-site dir for application logs.
sudo install -d -o ryan -g www-data -m 2775 ~/www/luna.hellyer.kiwi/public
sudo install -d -g www-data -m 2775 ~/logs/luna.hellyer.kiwi

# Placeholder PHP page.
echo '<?php echo "luna works";' | sudo tee ~/www/luna.hellyer.kiwi/public/index.php

# Re-apply ownership/permissions.
sudo bash scripts/fix-perms.sh ~/www/luna.hellyer.kiwi

# Validate the vhost and reload nginx.
sudo podman exec nginx nginx -t
sudo podman exec nginx nginx -s reload

# Expand the pressabl certificate to include luna.hellyer.kiwi, then reload.
# (Skips the cert if any listed domain does not resolve yet.)
sudo ./scripts/certbot-issue.sh

# Smoke test (validates against the real cert).
curl --resolve luna.hellyer.kiwi:443:127.0.0.1 https://luna.hellyer.kiwi/
